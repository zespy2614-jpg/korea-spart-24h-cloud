  .header-divider, #top-separator, .center-line {
    display: block;
    border-top: 1px solid #FFFFFF; 
    opacity: 0.5;
    margin: 5px auto;
  }

  display: none !important; /* !important를 사용하여 다른 스타일에 의해 재정의되는 것을 방지 */
